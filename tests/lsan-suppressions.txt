# Usage: ASAN_OPTIONS=... LSAN_OPTIONS=suppressions=tests/lsan-suppressions.txt

# (QV4::Heap::String::simplifyString via callWithInstance from
# DBusAdaptor::handleMessage — the frames ABOVE the leak are Qt's JS
# engine; no dbusqml allocation owns them). Fork-VM evidence:
# "Direct leak of 4193496 byte(s) in 12 object(s)" via
# QV4::RuntimeHelpers::addHelper <- CallQmlContextPropertyLookup <-
# QJSValue::callWithInstance. Qt-internal JS GC timing, not ours.
leak:QV4::Heap::String::simplifyString

# 2. QMetaObjectWrapper / ExecutionEngine::initializeGlobal /
